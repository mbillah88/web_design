document.getElementById("refundForm").addEventListener("submit", function (e) { e.preventDefault(); const formData = new FormData(this); fetch("/products/refund/", { method: "POST", body: formData }) .then(res => res.json()) .then(data => { if (data.success) { showToast("🔁 Product refunded successfully", "success"); bootstrap.Modal.getInstance(document.getElementById("refundModal")).hide(); resetRefundForm(); } else { showToast("❌ " + data.message, "danger"); } }) .catch(err => { console.error("Refund error:", err); showToast("❌ Unexpected refund error", "danger"); }); });
